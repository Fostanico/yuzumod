    }
}

# Yoshino's four lines exist in ordinary and bond-financed variants. The GUI
# deliberately reports shared stage numbers so both implementations remain one
# player-facing plan and no monthly polling is required.
defined_text = {
    name = yz_yoshino_industry_progress_panel
    text = {
        trigger = { has_country_flag = finished_yoshino_industry }
        localization_key = yz_yoshino_industry_progress_completed
    }
    text = {
        trigger = {
            NOT = {
                OR = {
                    has_country_flag = Yoshino_industry_five_year_yes
                    has_country_flag = Yoshino_industry_five_year_yes_by_loan
                }
            }
        }
        localization_key = yz_yoshino_industry_progress_not_started
    }
    text = { localization_key = yz_yoshino_industry_progress_active }
}
