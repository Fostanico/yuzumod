yz_academy_refresh_selected_leader_display = {
	ROOT = {
		clear_array = yz_academy_selected_leaders
		every_army_leader = {
			limit = { has_character_flag = have_been_chosen }
			ROOT = {
				add_to_array = {
					array = yz_academy_selected_leaders
					value = PREV
				}
			}
		}
		if = {
			limit = { check_variable = { yz_academy_selected_leaders^num < 1 } }
			every_army_leader = {
				limit = { has_trait = yz_training_trait }
				ROOT = {
					add_to_array = {
						array = yz_academy_selected_leaders
						value = PREV
					}
				}
			}
		}
	}
}
